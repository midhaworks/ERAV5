རྒྱ་གར། གཞུང་འབྲེལ་ཐོག་རྒྱ་གར་སྤྱི་མཐུན་རྒྱལ་ཁབ་(རྒྱ་གར་སྐད་དུ།: भारत गणराज्य) ཞེས་འབོད་པ་དང་། ལྷོ་ཨེ་ཤི་ཡའི་རྒྱལ་ཁབ། འཛམ་གླིང་གི་མི་འབོར་ཆེ་ཤོས་གཉིས་པ་ཡིན་པ་དང་། ས་རྒྱ་ཁྱོན་ཆ་ནས་འཛམ་གླིང་གི་ཨང་བདུན་པར་སླེབས་ཀྱི་ཡོད། མ་ཟད་དམངས་གཙོ་རྒྱལ་ཁབ་ཆེ་ཤོས་ཞིག་ཡིན། ཨེ་ཤི་ཡ་གླིང་ཆེན་འདིའི་ནང་ལོ་རྒྱུས་ཡུན་རིང་ལྡན་ཞིང་རིག་གནས་ཕུན་སུམ་ཚོགས་པའི་རྒྱལ་ཁབ་ཅིག་ཡིན། རྒྱ་གར་གྱི་སྐམ་ས་ཆེན་མོ་ནི་གོ་ལ་ཕྱེད་བྱང་མའི་ཕྱོགས་སུ་འཕྲེད་ཐིག་ 804’N ནས་ 3706’N ཡི་བར་དང་། གཞུང་ཐིག་ 6807’E ནས་ 97025’E ཡི་བར་དུ་ཆགས་ཡོད། རྒྱ་གར་གྱི་བྱང་ཕྱོགས་སུ་ཧི་མ་ལ་ཡའི་རི་རྒྱུད་དང་། ནུབ་ཕྱོགས་སུ་ཨ་རབ་རྒྱ་མཚོ། ཤར་ཕྱོགས་སུ་བངྒལ་གྱི་མཚོ་ཁུག ལྷོ་ཕྱོགས་སུ་རྒྱ་གར་རྒྱ་མཚོ་བཅས་ཀྱིས་བསྐོར་ཡོད། ཡུལ་འདིའི་ས་ཆའི་རྒྱ་ཁྱོན་ལ་སྤྱི་ལེ་གྲུ་བཞི་མ་ས་ཡ 3.28 ཡོད་པ་དང་། བྱང་གི་ཀཱསྨིར་མངའ་སྡེ་ནས་ལྷོ་ཡི་ཀཱན་ཡ་ཀཱུ་མ་རི་དབར་བར་ཐག་ཀི་ལོ་མི་ཊར་ 3,214 ཡོད། དེ་བཞིན་ཤར་གྱི་ཨ་རུ་ནཱ་ཅལ་མངའ་སྡེ་ནས་ནུབ་ཀྱི་གུཇ་རཱཏ་དབར་བར་ཐག་ཀི་ལོ་མི་ཊར་ 2,933 ཡོད། ས་ཆའི་ཆགས་སྟངས་ནི་དཔངས་མཐོ་བའི་རི་བོ་དང་རྒྱ་ཆེ་བའི་བྱེ་ཐང་། བྱང་ ཕྱོགས་གྱི་བདེ་ཐང་། ངོ་མཚར་ཆེ་བའི་མཐོ་སྒང། མཚོ་ཁ་དང་གླིང་ཕྲན་ སོགས་སྣ་མང་རང་བཞིན་གྱིས་གྲུབ་ཡོད། ས་ཁྱོན་ཆེ་བའི་རྒྱལ་ཁབ་འདིའི་གནམ་གཤིས་དང་རྩི་ཤིང་། སྲོག་ཆགས། རིག་གནས་སོགས་ནི་རྒྱལ་ཁབ་གཞན་དང་མི་འདྲ་བའི་ཁྱད་པར་ཆེན་པོ་ཡོད།

རྒྱ་གར་ལ་སྐམ་ས་དང་འབྲེལ་བའི་ཁྱིམ་མཚེས་རྒྱལ་ཁབ་བདུན་ཡོད་པ་དེ་དག་ནི། བོད་དང་བལ་ཡུལ། འབྲུག། བངྒ་ལ་དེཤ། འབར་མ། པཀི་སྟཱན། ཨཧྥ་ག་ནི་སྟཱན་བཅས་ཡིན། ད་དུང་རྒྱ་མཚོས་བར་ཆོད་ཅིང་ཕན་ཚུལ་ཁ་སྤྲོད་དུ་གནས་པའི་རྒྱལ་ཁབ་ཅིག་ཡོད་པ་དེ་ནི་ཤྲི་ལངྐ་ཡིན། རྒྱ་གར་གྱི་ཆ་ཤས་ ཨན་ཌ་མན་དང་ནཱི་ཀོ་རྦར་གླིང་ཕྲན་ནི་ཐེ་ལན་ཊ་དང་ཨིན་རྡོ་ནི་ཥི་ཡ་བར་མཚོ་ཐོག་རྒྱལ་མཚམས་འབྲེལ་ཡོད། པཀི་སྟཱན་དང་བངྒ་ལ་དེཤ་གཉིས་ནི་སྔར་རྒྱ་གར་གྱི་ཆ་ཤས་ཤིག་ཡིན་ཡང་ཕྱིས་སུ་ལོགས་སུ་གྱེས་ནས་རྒྱལ་ཁབ་མི་འདྲ་བ་གསུམ་དུ་གྱེས་པ་རེད།

རྒྱ་གར་ནི་སེང་གེ་གཙང་པོའི་གཤོངས་ཀྱི་དཔལ་ཡོན་དར་བའི་གནས་ཡིན་པ་དང་ད་ལྟའི་ཆར་པ་ཀི་སི་ཐན་ཁུལ་དུ་ཡོད་པའི་ཧར་པན་གནའ་རྫས་ལས་རྒྱ་སྐྱེད་དང་དབང་ཤུགས་སོགས་རྟོགས་ཐུབ། ཧིན་རྡུ་ཆོས་ལུགས་དང་། ནང་ཆོས། ཇིན་ཆོས་ལུགས། སིག་ཆོས་ལུགས་བཅས་ཀྱི་འབྱུང་ཁུངས་ཡིན། མ་ཟད་ ཇུ་ཌའི་ཆོས་ལུགས། ཛོ་རཱས་ཊི་ཆོས་ལུགས། ཡི་ཤུ་ཆོས་ལུགས། ཁ་ཆེའི་ཆོས་ལུགས་བཅས་སྤྱི་ལོའི་དུས་རབས་དང་པོའི་ནང་དར་ཁྱབ་བྱུང་། དེས་རྐྱེན་པས་སྣ་མང་རིག་གཞུང་ཕུན་སུམ་ཚོགས་པ་ཞིག་རྒྱ་གར་སྐམ་ས་ཆེན་མོ་ཁྱོན་དུ་དར་ཁྱབ་བྱུང་། ཕྱིས་སུ་རིམ་པས་ རྒྱ་གར་ནང་དབྱིན་ཇི་དབང་ཤུགས་སླེབས་ཏེ་མངའ་ཁོངས་ཡོངས་སུ་ཟླ་སྒྲིལ་བྱས། སྤྱི་ལོའི་དུས་རབས་བཅོ་བརྒྱད་ནས་བཅུ་དགུ་བར་ཤར་རྒྱ་གར་དབྱིན་ཇིའི་ཚོང་འབྲེལ་ཀུང་སིས་རྒྱ་གར་རྒྱལ་ཡོངས་འཛིན་སྐྱོང་བྱས་། བདག་ཉིད་ཆེན་པོ་གྷན་དྷི་མཆོག་གི་དབུ་ཁྲིད་འོག་དབྱིན་གཞུང་གི་བཀོད་ཁྱབ་མི་བརྩི་བའི་ཞི་རྒོལ་ལས་འགུལ་(འཆི་གསོན་དབྱེ་ཐག་གཅོད་པའི་ལས་འགུལ) གཞི་རྒྱ་ཆེ་བ་ཞིག་སྤེལ་བས། སྤྱི་ལོ་ 1947 ཟླ་ 8 ཚེས་ 14 ཡི་མཚན་དཀྱིལ་དུ་རྒྱ་གར་ལ་རང་བཙན་ཐོབ།

## ཐ་སྙད་བྱུང་ཚུལ།

རྒྱ་གར་འཕགས་པའི་ཡུལ་ཞེས་པ་དེ་ཡང་། བདག་ཅག་གི་སྟོན་པ་ཤཀྱ་ཐུབ་པ་དང་མཁས་གྲུབ་བླ་མེད་ཀླུ་སྒྲུབ་ཡབ་སྲས་སོགས་བཀའ་དྲིན་ཅན་གྱི་འཕགས་པའི་གང་ཟག་དུ་མ་ཞབས་བཅག་པའི་ཡུལ་ཡིན་པས་དེ་ལྟར་འབོད་རྒྱ་གར་ཞེས་པ་ནི་ཐག་རིང་གི་ཡུལ་གྲུ་དང་དེར་གནས་པའི་མི་རྣམས་ཀྱི་གོས་དཀར་པོ་གྱོན་སྲོལ་ཡོད་པས་སོ། དབྱིན་སྐད་དུ་ཨིན་རྡི་ཡ་ཞེས་འབོད་པ་དང་། དེ་ཡང་ཏ་ཟིག་ཡུལ་སྐད་དུ་སེང་གེ་གཙང་པོར་ཨིན་རྡུ་ས་ཞེས་འབོད་པ་ལས་བྱུང་། ལེགས་སྦྱར་སྐད་དུ་སཱིན་རྡུ་ཞེས་འབོད་པ་དེ་ཡིན། གནའ་རབས་ཀྱི་གྷི་རི་སི་སྐད་དུ་ཨིན་རྡུ་ཞེས་འབོད་པ་དང་། དོན་དུ་སེང་གེ་གཙང་པོའི་ཉེ་གདྭབས་སུ་གནས་པའི་མི་ལ་ཟེར།

རྒྱ་གར་བ་རང་ཉིད་ནས་བྷར་ར་ཊ་ཞེས་འབོད་ཀྱི་ཡོད་པ་དང་། རྩ་ཁྲིམས་ནས་གཞུང་འབྲེལ་ཀྱི་མིང་དུ་ངོས་འཛིན་དུ་གནང་ཡོད། རྒྱ་གར་སྐད་རིགས་འདྲ་མིན་ནང་བེད་སྤྱོད་བྱེད་ཀྱི་ཡོད། ཧིན་རྡུ་ཆོས་ལུགས་ནང་ངོས་འཛིན་བྱེད་པའི་རྒྱ་གར་གྱི་གནའ་རབས་རྒྱལ་པོ་གྲགས་ཅན་ཞིག་ཀྱི་མིང་ཡིན།

མིང་གཞན་ཞིག་དུ་ཧིན་རྡུ་སི་ཐན་ཞེས་ཀྱང་འབོད། པཱར་ཤན་སྐད་དུ་ཧིན་རྡུ་ཆོས་ལུགས་ཀྱི་རྗེས་འབྲངས་པ་གནས་པའི་ཡུལ་ལ་ཟེར།

## འཛིན་སྐྱོང་མངའ་ཁོངས།

རྒྱ་གར་ནི་ས་ཁྱོན་ཧ་ཅང་ཆེ་ཞིང་སྐད་ཡིག་མང་པོ་ཡོད་པའི་རྒྱལ་ཁབ་ཅིག་ཡིན་པས། ཡུལ་མིའི་ཁེ་དབང་སྲུང་སྐྱོབ་དང་འཛིན་སྐྱོང་ཚགས་ཚུད། ལས་དོན་མཉམ་སྒྲུབ། བདེ་སྐྱིད་མཉམ་མྱོང་སོགས་ཀྱི་ཆེད་རྒྱལ་ཁབ་ཧྲིལ་པོ་དེ་མངའ་སྡེ་ 28 དང་ཐད་སྐྱོང་ས་ཁུལ་ 7 ལ་བགོས་ཡོད། ས་ཁྱོན་གྱི་ཆ་ནས་རཱ་ཇ་སྠཱན་ནི་མངའ་སྡེ་ཆེ་ཤོས་དང་། ཆུང་ཤོས་ནི་གྷོ་བ་(Goa)རེད། མངའ་སྡེ་དེ་དག་ཀྱང་རྫོང་དང་གྲོང་གསེབ་རིམ་པར་དབྱེ་ནས་འཛིན་སྐྱོང་བྱེད་བཞིན་ཡོད། ད་ལྟའི་ཆར་དབུས་གཞུང་ནི་རྒྱལ་ས་དིལླིར་ཆགས་ཡོད།

## ས་ཁམས་ཆགས་ཚུལ།

རྒྱ་གར་ནི་རི་བོ་དང་མཐོ་སྒང་། གཤོངས་ས། བདེ་ཐང་། མཚོ་ཁ། གླིང་ཕྲན་སོགས་ས་དབྱིབས་དུ་མས་གྲུབ་ཡོད་པ་ལས། ཧི་མ་ལ་ཡ་དང་། གངྒཱའི་བདེ་ཐང་། ཌེཀྐན་མཐོ་སྒང་བཅས་དབྱེ་བགོད་གསུམ་དུའང་ཕྱེ་ཆོག་པ་ཡིན། དེ་ལས་ཧི་མ་ལ་ཡ་ཞེས་པ་ནི་ལེགས་སྦྱར་གྱི་སྐད་དང་བོད་སྐད་དུ་རི་བོ་གངས་ཅན་ཞེས་བརྗོད་པ་ཡིན། རི་བོ་འདི་ནི་འཛམ་གླིང་གི་རི་བོ་མཐོ་ཤོས་དང་རི་རྒྱུད་རིང་ཤོས་ཤིག་ཡིན། ཧི་མ་ལ་ཡ་ནི་ཧི་མ་དི་རི་དང་། ཧི་མ་ཅལ། ཤི་ཝ་ལི་བཅས་རི་རྒྱུད་གསུམ་གྱིས་གྲུབ། རྒྱ་གར་བྱང་ཕྱོགས་བདེ་ཐང་ནི་ཧི་མ་ལ་ཡའི་ལྷོ་ཕྱོགས་སུ་ཆགས་ཡོད། གངྒཱ་དང་ཧིན་དྷུ་གཙང་པོ།(སེང་གེ་ཁ་འབབ) བྲ་མཧཱ་པུཏྲ་(རྟ་མཆོག་ཁ་འབབ་)བཅས་དང་དེ་དག་གི་ཆུ་ལག་གིས་བྱེ་འདམ་རིམ་གསོག་གིས་བདེ་ཐང་དེ་ཆགས་པ་ཡིན། བདེ་ཐང་འདི་རྒྱ་ཆེ་ལ་ངོས་སྙོམས་པ་མ་ཟད། ས་གཤིས་གཤིན་པོ་ཡོད་ལ་ཞིང་ལས་ཐོན་སྐྱེད་བཟང་བས། ས་ཁུལ་འདིར་མི་ཚོགས་ཀྱང་མང་པོ་འདུས་སྡོད་བྱས་ཡོད། རྒྱ་གར་གྱི་ནུབ་ཕྱོགས་སུ་རྒྱ་གར་བྱེ་ཐང་ཆེན་མོའམ་ཐཱར་བྱེ་ཐང་ (Great Indian desert) ཆགས་ཡོད། ཁུལ་དེར་སྐམ་ཤས་དང་ཚ་དྲོད་ཆེ་བས་རྩི་ཤིང་ཉུང་ངུ་ལས་སྐྱེ་ཐུབ་ཀྱི་མེད། རྒྱ་གར་གྱི་ལྷོ་ཕྱོགས་སུ་ཕྱེད་གླིང་མཐོ་སྒང་(Peninsular plateau) ཡོད། དེའི་དབྱིབས་ཟུར་གསུམ་ཡིན་པས་ས་དབྱིབས་གཞན་དང་མི་འདྲ་བའི་ཁྱད་པར་ཆེན་པོ་མཐོང་ཐུབ། ཁུལ་འདིར་ཨ་རཱ་ཝ་ལི་རི་(Aravali hill) དང་། ཝིན་དི་ཡཱ་ (Vindhyas) དང་སཱ་ཏེ་པུ་ར་ (Satpuras) སོགས་ཀྱི་རི་རྒྱུད། ནརྨ་དཱ་ (Narmada) དང་ཏཱ་པི་ (Tapi) གཙང་པོ་སོགས་ཆགས་ཡོད་ཅིང་། གཙང་པོ་འདི་དག་ནུབ་ཕྱོགས་སུ་བཞུར་ནས་ཨ་རབ་རྒྱ་མཚོའི་ནང་འབབ་ཀྱི་ཡོད། བྱང་རི་ (Western Ghats) སཱ་ཡ་དི་རིའི་རི་རྒྱུད་ཀྱིས་བྱང་གི་ས་མཚམས་བྱེད་པ་དང་། ཤར་རིའི་ (Eastern Ghats) རི་རྒྱུད་ཀྱིས་ཤར་གྱི་ས་མཚམས་མཚོན་པར་བྱེད། རི་རྒྱུད་འདི་གཉིས་ཀྱི་འདབས་སུ་མཚོ་ཁའི་བདེ་ཐང་ (Coastal Plain) ཆགས་ཡོད་ཅིང་། ཁུལ་འདིར་རྡོ་སོལ་དང་ལྕགས་རིགས་སོགས་གཏེར་ཁའི་ཐོན་ཁུངས་ལེགས་པོ་ཡོད། མ་ཧཱ་ན་དི་དང་། (Mahanadi) གོ་དཱ་ཝ་རཱི། (Godavari) ཀྲྀཥྞཱ (Krishna) ཀཱ་ཝེ་རཱི་ (Kaveri) གཙང་པོ་སོགས་ནི་ཤར་ཕྱོགས་སུ་བཞུར་ནས་མཐར་བངྒལ་མཚོ་ཁུག་ཏུ་འབབ་ཀྱི་ཡོད། སྐམ་ས་ཆེན་མོ་ཕུད་ད་དུང་རྒྱ་གར་གྱི་མངའ་ཁོངས་སུ་གླིང་ཕྲན་གཉིས་ཡོད་པ་ལས། ཨ་རབ་རྒྱ་མཚོའི་ངོགས་སུ་ལཀྵ་དཱིབ་གླིང་ཕྲན་ (Lakshadweep Islands) དང་། བངྒལ་མཚོ་ཁུག་ཏུ་ཨནྜ་མཱན་དང་ནི་ཀོ་བཱར་ (An- daman and Nicobar Islands) གླིང་ཕྲན་ཆགས་ཡོད།

## སྐྱེས་དངོས་དང་ཁོར་ཡུག།

རྒྱ་གར་ལ་རྩྭ་དང་རྩི་ཤིང་སྣ་ཚོགས་ཡོད་ཀྱང་། དེ་དག་རྩ་བའི་རིགས་ལྔར་དབྱེ་བ་སྟེ། ཚ་ཁུལ་གྱི་རྒྱུན་ལྗང་ནགས་ཚལ་(Tropical evergreen forest) དང་། ཚ་ཁུལ་གྱི་ལོ་མ་དུས་སུ་ལྷུང་བའི་ནགས་ཚལ། (Tropi- cal deciduous forest) ཚེར་ཤིང་།(Thorny bushes) རི་ཤིང་། (Mountain vegetation) མཚོ་ཁའི་ཆུ་ཤིང་ (Mangrove forest) བཅས་སོ།། ནགས་ཚལ་ནི་རི་སྐྱེས་སྲོག་ཆགས་མང་པོ་འཚོ་གནས་བྱེད་ས་ཡིན། མིས་བདག་ཏུ་མ་བཟུང་བའི་སྲོག་ཆགས་ ལ་རི་སྐྱེས་སྲོག་ཆགས་ཟེར། རྒྱ་གར་ལ་ལྟོ་འགྲོའི་སྲོག་ཆགས་(Reptiles)དང་། ས་ཆུ་གཉིས་སྡོད་ཀྱི་སྲོག་ཆགས།(Amphibians) འོ་འཐུང་སྲོག་ཆགས། (Mammals) བྱ་རིགས། འབུ་སྲིན་སོགས་སྲོག་ཆགས་རིགས་སྣ་ཚོགས་ཡོད། ཤིང་ནགས་གང་བྱུང་དུ་གཅོད་འབྲེག་དང་། རྨ་བྱ་དང་ཁྲུང་ཁྲུང་སོགས་བྱ་རིགས་དང་། སྟག་དང་། གཟིག གླང་ཆེན་ སོགས་རྩ་ཆེའི་སྲོག་ཆགས་རིགས་རྔོན་རྒྱག་བྱས་དང་བྱེད་མུས་ལ་བརྟེན། དེང་སྐབས་རི་སྐྱེས་སྲོག་ཆགས་མང་པོ་ཞིག་རྩ་སྟོང་འགྲོ་བའི་ཉེན་ཁ་ཆེན་པོ་འཕྲད་ཀྱིན་ཡོད། རྩ་སྟོང་འགྲོ་ཉེན་ཆེ་བའི་སྲོག་ཆགས་དེ་དག་སྲུང་སྐྱོབ་བྱ་ཆེད་གཅན་གཟན་ཁང་དང་། གླིང་ག་སོགས་བཟོ་སྐྲུན་གང་འཚམ་བྱས་ཡོད་ནའང་། རི་སྐྱེས་སྲོག་ཆགས་རྣམས་རྩ་སྟོང་འགྲོ་བའི་ཉེན་ཁ་མུ་མཐུད་འཕྲད་བཞིན་ཡོད།

## དཔལ་འབྱོར།

རྒྱ་གར་མིག་སྔར་འཕེལ་རྒྱས་སུ་ཕྱིན་ཀྱིན་པའི་འབྱོར་ཞན་རྒྱལ་ཁབ་ཅིག་ཡིན། ཡིན་ནའང་སློབ་གསོ་ཡི་སྤུས་ཀ་བཟང་བ་དང་མི་འབོར་གྱི་རྒྱབ་བརྟེན་བཟང་བས་རྐྱེན་གྱིས་ད་སྔ་འཛམ་གླིང་ཧྲིལ་བོ་ན་དཔལ་འབྱོར་ཐད་ནས་ཤུགས་རྐྱེན་ཆེན་པོ་འབྱུང་བཞིན་ཡོད་པའི་རྒྱལ་ཁབ་ཅིག་རེད། རྒྱ་གར་ནི་ཞིང་ལས་གཙོ་བའི་རྒྱལ་ཁབ་ཡིན་དང་། བཟོ་ལས་ད་ལྟ་མང་ཆེ་ཤོས་གྲོང་ཁྱེར་ཆེན་པོའི་ཉེ་འགྲམ་ན་འཕེལ་རྒྱས་ཆེན་པོ་བྱུང་ཡོད། ད་དུང་རྒྱ་གར་ནི་མཉེན་ཆས་ལས་རིགས་ཀྱི་རྟེན་ས་གཙོ་བོ་ཡིན་པ་དང་། རྒྱལ་ཁབ་དེ་ནས་བྱུང་བའི་མཉེན་ཆས་ལས་བཟོ་བས་འཛམ་གླིང་ཁྱོན་ཡོངས་ལ་ཤུགས་རྐྱེན་ཆེན་པོ་ཐེབས་ཀྱིན་ཡོད། ད་དུང་དབྱིན་ཡིག་གི་རྨང་གཞི་བཟང་བ་དང་ཞབས་ཞུའི་ལས་རིགས་ལ་ཞུགས་མཁན་གྱི་མི་འབོར་གྲངས་ཆེ་བས་ཞབས་ཞུ་ཁ་པར་དང་ལས་གཞི་གཞན་གླས་ཀྱི་ལས་ཀ་གཉེར་མཁན་ཤིན་ཏུ་མང་། ཡིན་ནའང་དཔལ་འབྱོར་གྱི་ཆ་ནས་འཕེལ་རྒྱས་ཆེ་ཞིང་གྲངས་འབོར་མཐོ་རུང་། འཛད་སྤྱོད་པའི་ཁ་གྲངས་བཟང་བ་དང་དབུག་ཕྱུག་ཆ་སྙོམས་མིན་པའི་གནད་དོན་ཤིན་ཏུ་ཛ་དྲག་ཡིན་སྟབས་ད་སྔ་རྒྱ་གར་གྱི་དཔལ་འབྱོར་ཐད་ལ་གནད་དོན་གང་མང་བྱུང་བཞིན་ཡོད་པ་རེད།

## མི་འབོར།

རྒྱ་གར་ནི་འཛམ་གླིང་སྟེང་གི་མི་འབོར་གྱི་གྲངས་ཀ་ཐེར་འབུམ་ལ་སོན་ཡོད་པའི་རྒྱལ་ཁབ་གཉིས་པ་ཡིན་ལ། མིག་སྔར་མི་འབོར་དུང་ཕྱུར་བཅུ་གཉིས་དང་བྱེ་བ་བརྒྱད་ལྷག་ཡོད། སྔོན་དཔོག་བྱེད་པ་ལྟར་ན། མི་ལོ་ལྔ་ནས་བཅུ་བར་ལ་རྒྱ་གར་གྱི་མི་འབོར་ཡོངས་སུ་རྒྱ་ནག་ལས་བརྒལ་ནས་འཛམ་གླིང་ཡོངས་ཀྱི་མི་འབོར་ཆེས་མང་བའི་རྒྱལ་ཁབ་དེ་རུ་གྱུར་ངེས་རེད།

## ཆོས་ལུགས།

རྒྱ་གར་ནི་འཛམ་གླིང་སྟེང་གི་ཆོས་ལུགས་དད་མོས་བྱེད་མཁན་དང་ཆོས་ལུགས་ཀྱི་རིགས། ད་དུང་ཆོས་ལུགས་ཀྱི་འཚོ་བར་ཤུགས་རྐྱེན་ཐེབས་ཡོད་པ་སོགས་གང་ཅི་ཐད་ནས་རྩེ་ལ་སོན་ཡོད་པའི་རྒྱལ་ཁབ་ཅིག་ཡིན། རྒྱ་གར་ལ་གཙོ་བོ་ཧིན་རྡུལ་ཆོས་ལུགས་སམ་བྲམ་ཟེ་ཆོས་ལུགས་ལ་དད་མོས་བྱེད་མཁན་མང་པོ་ཡོད་པ་ལས་གཞན། ད་དུང་ཁ་ཆེ་ཆོས་ལུགས་སམ་ཡེ་ཟི་ལམ་ཆོས་ལུགས། གནམ་བདག་ཆོས་ལུགས་དང་ཡེ་ཤེས་ཆོས་ལུགས། ད་དུང་ཇེ་ན་ཆོས་ལུགས་སོགས་རིགས་མང་པོ་ཡོད། དེ་བས་མ་ཟད་ལུང་བ་སོ་སོ་ན་རང་རང་གིས་རྟེན་གཙོ་དང་གཙོ་སྐུ་ཡོད་པ་དེ་ལ་དད་མོས་བྱེད་པ་ཡང་མང་དུ་མཐོང་། དེ་ཁྲོད་དུ་རྒྱལ་བོ་རཱ་མ་ནའི་སྒྲུང་ལ་འབྲེལ་བའི་གཏམ་རྒྱུད་དང་། ད་དུང་དམངས་ཁྲོད་དུ་གྲགས་པའི་རྒྱལ་བོ་སོགས་ལ་དད་པའི་ཆོས་ཡང་ཤན་ཏུ་མང་པོ་ཡོད་པ་རེད། ངེས་པར་གསལ་ཁ་འདོན་རྒྱུ་ཞིག་ལ། སངས་རྒྱས་ཆོས་ལུགས་ནི་སྤྱིར་རྒྱ་གར་དང་བལ་བོ་གཉིས་ཀྱི་ས་མཚམས་ནས་བྱུང་ན་ཡང་། ད་ལྟ་རྒྱ་གར་གྱི་སྤྱི་དམངས་ཁྲོད་ན་དེར་དད་པ་བྱེད་མཁན་གྱི་མི་གྲངས་སྤྱིའི་བརྒྱ་ཆ་དང་བསྡུར་ན་ཤིན་ཏུ་ཉུང་བའི་ཚོད་ལ་སླེབས་ཡོད་པ་རེད།
